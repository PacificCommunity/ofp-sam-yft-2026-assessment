\begin{frame}{Model development}\fns
  \vspace{-2ex}
  \begin{columns}[T]
    \column{0.45\textwidth}
    \textblue{Spatial structure}\\[-0.8ex]
    \begin{enumerate}\setlength\itemsep{-0.3ex}
      \item Consider single-region or equatorial model
      \item Revise regional structure (esp. bigeye)\\[1.5ex]
    \end{enumerate}
    \textblue{Fisheries}\\[-0.8ex]
    \begin{enumerate}\setcounter{enumi}{2}\setlength\itemsep{-0.3ex}
      \item Convert weight comps to length comps
      \item Combine fisheries
      \item Lower effective sample size in general
      \item Analyze data conflict, identify and test different
      approaches
      \item Possibly reduce sample size in regions without otolith data
      \item Consider different scenarios for Indonesian catches
      \item Consider using size composition from predominant nation in some
      fisheries
    \end{enumerate}
    \textblue{Recruitment}\\[-0.8ex]
    \begin{enumerate}\setcounter{enumi}{9}\setlength\itemsep{-0.3ex}
      \item Orthogonal polynomial recruitment\\[1.2ex]
    \end{enumerate}
    \column{0.45\textwidth}
    \textblue{Natural mortality}\\[-0.8ex]
    \begin{enumerate}\setcounter{enumi}{10}\setlength\itemsep{-0.3ex}
      \item Fix at different levels of M\\[1.2ex]
    \end{enumerate}
    \textblue{Movement}\\[-0.8ex]
    \begin{enumerate}\setcounter{enumi}{11}\setlength\itemsep{-0.3ex}
      \item SEAPODYM coefficients\\[1.2ex]
    \end{enumerate}
    \textblue{Growth}\\[-0.8ex]
    \begin{enumerate}\setcounter{enumi}{12}\setlength\itemsep{-0.3ex}
      \item Reweight otoliths based on location
      \item Richards curve\\[1.2ex]
    \end{enumerate}
    \textblue{Selectivity}\\[-0.8ex]
    \begin{enumerate}\setcounter{enumi}{14}\setlength\itemsep{-0.3ex}
      \item Length-based\\[1.2ex]
    \end{enumerate}
    \textblue{Tags}\\[-0.8ex]
    \begin{enumerate}\setcounter{enumi}{15}\setlength\itemsep{-0.3ex}
      \item Mixing period for each release group\\[1.2ex]
    \end{enumerate}
    \textblue{CPUE}\\[-0.8ex]
    \begin{enumerate}\setcounter{enumi}{16}\setlength\itemsep{-0.3ex}
      \item Alternative indices
      \item Time-varying CV
      \item Effort creep
    \end{enumerate}
  \end{columns}
